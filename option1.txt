    <script>
        function initiateSync() {
            const uName = document.getElementById('user-name').value.trim();
            const aName = document.getElementById('ai-name').value.trim();
            const aModel = document.getElementById('ai-model').value;

            if (!uName || !aName) {
                alert("Both names are required to initiate sync.");
                return;
            }

            // Save data to localStorage to use in chat1.html
            localStorage.setItem('trinova_m1_user', uName);
            localStorage.setItem('trinova_m1_ai', aName);
            localStorage.setItem('trinova_m1_model', aModel);

            // Redirect to chat screen
            window.location.href = 'chat1.html';
        }
    </script>
