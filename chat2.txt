    <script>
        // LocalStorage se details uthana
        const ai1Name = localStorage.getItem('trinova_m2_ai1_name');
        const ai1Model = localStorage.getItem('trinova_m2_ai1_model');
        const ai2Name = localStorage.getItem('trinova_m2_ai2_name');
        const ai2Model = localStorage.getItem('trinova_m2_ai2_model');
        const debateTopic = localStorage.getItem('trinova_m2_topic');

        // Agar user direct page par aa gaya, wapas bhej do
        if (!ai1Name || !ai2Name || !debateTopic) {
            window.location.href = 'option2.html';
        }

        // State Variables
        let currentTurn = 1; // 1 means AI 1 ki baari, 2 means AI 2 ki baari
        let lastMessage = ""; // Dusre AI ne kya bola, yaad rakhne ke liye

        // Page load hone par
        window.onload = () => {
            document.getElementById('vs-text').innerText = `${ai1Name} (${ai1Model}) VS ${ai2Name} (${ai2Model})`;
            appendMessage('SYSTEM', `Topic: "${debateTopic}"`, 'system');
        };

        // Asli Behas (Debate) Logic
        async function generateNextReply() {
            const btn = document.getElementById('next-btn');
            const typingIndicator = document.getElementById('typing-indicator');

            // Set current AI info
            let activeName, activeModel, otherName;
            if (currentTurn === 1) {
                activeName = ai1Name; activeModel = ai1Model; otherName = ai2Name;
            } else {
                activeName = ai2Name; activeModel = ai2Model; otherName = ai1Name;
            }

            // UI update
            btn.disabled = true;
            btn.innerText = "Processing...";
            typingIndicator.innerText = `${activeName} is typing...`;
            typingIndicator.style.display = 'block';
            scrollToBottom();

            // Background Context (API ko pata hona chahiye kis baare mein behas karni hai)
            let promptMessage = "";
            if (lastMessage === "") {
                promptMessage = `The debate topic is: "${debateTopic}". Give your opening argument in 2-3 lines.`;
            } else {
                promptMessage = `The debate topic is: "${debateTopic}". ${otherName} just said: "${lastMessage}". Give a strong counter-reply to ${otherName} in 2-3 lines. Do not agree with them easily.`;
            }

            try {
                const response = await fetch('api.php', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        mode: 'option2',
                        model: activeModel,
                        ai_name: activeName,
                        message: promptMessage
                    })
                });
                
                const data = await response.json();
                
                typingIndicator.style.display = 'none';
                btn.disabled = false;
                btn.innerText = "GET NEXT REPLY 🥊";

                if(data.reply) {
                    // Screen par message dikhao
                    appendMessage(activeName, data.reply, currentTurn === 1 ? 'ai1' : 'ai2');
                    
                    // Context update karo
                    lastMessage = data.reply;
                    
                    // Turn change karo
                    currentTurn = currentTurn === 1 ? 2 : 1;
                } else {
                    appendMessage('System', 'Error: No reply received.', 'system');
                }

            } catch (error) {
                typingIndicator.style.display = 'none';
                btn.disabled = false;
                btn.innerText = "GET NEXT REPLY 🥊";
                appendMessage('System', 'Server connection failed!', 'system');
            }
        }

        // Message Append Logic
        function appendMessage(senderName, text, type) {
            const chatBox = document.getElementById('chat-box');
            
            const wrapperDiv = document.createElement('div');
            wrapperDiv.className = `msg-wrapper ${type}-wrapper`; // class e.g. ai1-wrapper
            
            if (type !== 'system') {
                const senderSpan = document.createElement('span');
                senderSpan.className = 'msg-sender';
                senderSpan.innerText = senderName;
                wrapperDiv.appendChild(senderSpan);
            }
            
            const msgDiv = document.createElement('div');
            msgDiv.className = 'msg';
            msgDiv.innerText = text;
            
            wrapperDiv.appendChild(msgDiv);
            chatBox.appendChild(wrapperDiv);
            
            scrollToBottom();
        }

        function scrollToBottom() {
            const chatBox = document.getElementById('chat-box');
            chatBox.scrollTop = chatBox.scrollHeight;
        }
    </script>
